<%*
// 通常儲存在 SlimefunItemSetup.java
// https://github.com/Slimefun/Slimefun4/blob/experimental/src/main/java/io/github/thebusybiscuit/slimefun4/implementation/setup/SlimefunItemSetup.java
// 1. 彈出視窗詢問
let rawInput = await tp.system.prompt("請輸入新物品名稱 (若有別名請用空格隔開)：");
let itemId = await tp.system.prompt("請輸入物品 ID (例如: HM_QKJCY)：");

// 防呆機制
if (!rawInput || !itemId) return;

let nameParts = rawInput.trim().split(/\s+/);
let itemName = nameParts[0]; 
let itemAliases = nameParts.slice(1); 

// ==========================================
// 2 & 3. 跨檔案自動抓取：研究經驗 (levelCost) 與 Java 配方源碼
// ==========================================
let levelCost = "";
let itemSource = "";
let itemGroupRaw = "";
let outputQty = 1;
let outputExplicit = false;

const referenceFolder = "參考資料";

try {
    // 直接讀取「參考資料」資料夾
    const list = await app.vault.adapter.list(referenceFolder);

    // Slimefun ID 基本上只使用英文大寫、數字與底線，
    // 直接拿來建立搜尋字串即可。
    for (let filePath of list.files) {

        // Java 原始碼；也保留 txt / md，方便你把原碼貼成文字檔
        if (
            filePath.endsWith(".java") ||
            filePath.endsWith(".txt") ||
            filePath.endsWith(".md")
        ) {
            let content = await app.vault.adapter.read(filePath);

            // ======================================
            // 任務 A：找研究經驗
            //
            // 範例：
            // register(
            //   "monster_jerky",
            //   13,
            //   "Monster Jerky",
            //   5,
            //   SlimefunItems.MONSTER_JERKY
            // );
            //
            // 第四個參數 5 = levelCost
            // ======================================
            if (!levelCost) {

                let researchBlocks =
                    content.match(/register\([\s\S]*?\);/g) || [];

                let targetResearchRegex =
                    new RegExp(
                        "SlimefunItems\\." + itemId + "\\b"
                    );

                for (let block of researchBlocks) {

                    if (!targetResearchRegex.test(block)) {
                        continue;
                    }

                    let levelMatch =
                        block.match(
                            /register\(\s*"[^"]*"\s*,\s*\d+\s*,\s*"[^"]*"\s*,\s*(\d+)/
                        );

                    if (levelMatch) {
                        levelCost = levelMatch[1];
                        break;
                    }
                }
            }

            // ======================================
            // 任務 B：找物品註冊 / 配方區塊
            //
            // 每個物品通常最後以：
            // .register(plugin);
            // 結束，所以直接把 Java 原碼按註冊區塊切開。
            // ======================================
if (!itemSource) {

    let itemBlocks =
        content.match(
            /[\s\S]*?\.register\(plugin\);/g
        ) || [];

    let escapedItemId =
        itemId.replace(
            /[.*+?^${}()|[\]\\]/g,
            "\\$&"
        );


    // 必須是：
    //
    // new XXX(
    //     itemGroups.xxx,
    //     SlimefunItems.GOLD_DUST,
    //
    // 才算 GOLD_DUST 自己的註冊
    //
    // 如果 GOLD_DUST 只出現在 recipe 裡，
    // 不會誤判。
    let targetRegistrationRegex =
        new RegExp(
            "\\bnew\\s+[A-Za-z0-9_]+\\s*\\(" +
            "\\s*itemGroups\\.[A-Za-z0-9_]+\\s*," +
            "\\s*SlimefunItems\\." +
            escapedItemId +
            "\\b"
        );


    for (let block of itemBlocks) {

        if (
            targetRegistrationRegex.test(block)
        ) {
            itemSource = block;
            break;
        }
    }
}

            // 兩個都找到了就停止掃描
            if (levelCost && itemSource) {
                break;
            }
        }
    }

} catch (error) {

    new Notice(
        `⚠️ 搜尋參考資料時發生錯誤：${error.message}`
    );
}
// ======================================
// 解析明確產出數量
// ======================================
let escapedOutputId =
    itemId.replace(
        /[.*+?^${}()|[\]\\]/g,
        "\\$&"
    );

let outputRegex =
    new RegExp(
        "new\\s+ItemStack\\[\\]\\s*\\{" +
        "[\\s\\S]*?\\}" +
        "\\s*,\\s*" +
        "new\\s+SlimefunItemStack\\(" +
        "\\s*SlimefunItems\\." +
        escapedOutputId +
        "\\s*,\\s*(\\d+)\\s*" +
        "\\)\\.item\\(\\)"
    );

let outputMatch =
    itemSource.match(outputRegex);

if (outputMatch) {
    outputQty = Number(outputMatch[1]);
    outputExplicit = true;
}
// ==========================================
// 4. 解析取得的 Java 配方源碼
// ==========================================
let machine = "無/手動";
let machineInDict = false;
let materials = {};

let capacity = null;
let energyPerSecond = null;
let tickRate = null;

if (itemSource) {

    // ======================================
    // A. ItemGroup
    //
    // itemGroups.food → food
    // ======================================
    let itemGroupMatch =
        itemSource.match(
            /itemGroups\.([A-Za-z0-9_]+)/
        );

    if (itemGroupMatch) {
        itemGroupRaw = itemGroupMatch[1];
    }




    // ======================================
    // B. 生產設備 RecipeType
    // ======================================
    const machineDict = {
        "ENHANCED_CRAFTING_TABLE": "強化工作台",
        "MAGIC_WORKBENCH": "魔法工作台",
        "ARMOR_FORGE": "護甲鍛造台",
        "SMELTERY": "冶煉爐",
        "ORE_CRUSHER": "碎礦機",
        "ORE_WASHER": "洗礦機",
        "COMPRESSOR": "壓縮機",
        "JUICER": "榨汁機",
        "ANCIENT_ALTAR": "遠古祭壇",
        "PRESSURE_CHAMBER": "壓力艙",
        "GRIND_STONE": "磨石",
        "GOLD_PAN": "淘金盤",
        "HEATED_PRESSURE_CHAMBER": "加熱壓力艙"
    };

    let machineMatch =
        itemSource.match(
            /RecipeType\.([A-Za-z0-9_]+)/
        );

    let machineRaw =
        machineMatch
            ? machineMatch[1]
            : "無/手動";

    machineInDict =
        machineDict[machineRaw] !== undefined;

    machine =
        machineDict[machineRaw] || machineRaw;


    // ======================================
    // C. 配方材料
    //
    // 範例：
    // new ItemStack[] {
    //   SlimefunItems.SALT.item(),
    //   new ItemStack(Material.COOKED_BEEF),
    //   null ...
    // }
    // ======================================
    let recipeMatch =
        itemSource.match(
            /new\s+ItemStack\[\]\s*\{([\s\S]*?)\}/
        );

    if (recipeMatch) {

        let recipeSource = recipeMatch[1];
        let match;


        // ----------------------------------
        // C-1. SlimefunItemStack + 指定數量
        //
        // new SlimefunItemStack(
        //     SlimefunItems.STEEL_INGOT,
        //     8
        // ).item()
        // ----------------------------------
        const sfAmountRegex =
            /new\s+SlimefunItemStack\(\s*SlimefunItems\.([A-Za-z0-9_]+)\s*,\s*(\d+)\s*\)\.item\(\)/g;

        while (
            (match = sfAmountRegex.exec(recipeSource)) !== null
        ) {
            let material = match[1];
            let amount = Number(match[2]);

            if (!materials[material]) {
                materials[material] = 0;
            }

            materials[material] += amount;
        }


        // ----------------------------------
        // C-2. 一般 Slimefun 物品
        //
        // SlimefunItems.SALT.item()
        // ----------------------------------
        const sfItemRegex =
            /SlimefunItems\.([A-Za-z0-9_]+)\.item\(\)/g;

        while (
            (match = sfItemRegex.exec(recipeSource)) !== null
        ) {
            let material = match[1];

            if (!materials[material]) {
                materials[material] = 0;
            }

            materials[material] += 1;
        }


        // ----------------------------------
        // C-3. Minecraft 原版物品
        //
        // new ItemStack(Material.COOKED_BEEF)
        // new ItemStack(Material.IRON_INGOT, 4)
        // ----------------------------------
        const vanillaRegex =
            /new\s+ItemStack\(\s*Material\.([A-Za-z0-9_]+)(?:\s*,\s*(\d+))?\s*\)/g;

        while (
            (match = vanillaRegex.exec(recipeSource)) !== null
        ) {
            let material = match[1];
            let amount =
                match[2]
                    ? Number(match[2])
                    : 1;

            if (!materials[material]) {
                materials[material] = 0;
            }

            materials[material] += amount;
        }
    }
}


// 抓不到 output 時只提醒，不自行填 1
if (!outputExplicit) {
    new Notice(
        `⚠️ ${itemId} 未找到明確 output 數量，目前預設為 1，請再次確認`
    );
}

// ========================================
// 5. 建立「材料 ID → 現有檔案名稱」翻譯字典
// ========================================
const files = app.vault.getMarkdownFiles();
const typeDict = {
    "sf-item": "item",
    "sf-GEO": "GEO",
    "sf-drop": "drop"
};
const aliasToName = {};
const aliasToType = {};

for (let f of files) {

    const base = f.basename;

    const frontmatter =
        app.metadataCache.getFileCache(f)?.frontmatter || {};

    let aliases = frontmatter.aliases || [];
    let fileType = frontmatter.type || "sf-item";

    if (!Array.isArray(aliases)) {
        aliases = [aliases];
    }

    // sf-item → item
    // sf-GEO  → GEO
    // sf-drop → drop
    const outputType =
        typeDict[fileType] || "item";

    aliasToName[base] = base;
    aliasToType[base] = outputType;

    for (let alias of aliases) {

        if (alias) {

            const key = String(alias).trim();

            aliasToName[key] = base;
            aliasToType[key] = outputType;
        }
    }
}


// ========================================
// 6. 組合配方清單
// ========================================
let recipeOutput = "";
const DV = "::";


// ======================================
// 生產設備
// ======================================
if (machineInDict) {
    recipeOutput +=
        `- **生產設備**：(Machine${DV} [[${machine}]])\n`;
} else {
    recipeOutput +=
        `- **生產設備**：(Machine${DV} ${machine})\n`;
}


// ======================================
// 機器資訊：有才顯示
// ======================================
if (energyPerSecond !== null) {
    recipeOutput +=
        `\t- **耗電量**：[energy${DV} ${energyPerSecond}] J/s\n`;
}

if (tickRate !== null) {

    let intervalSeconds =
        tickRate * 0.5;

    recipeOutput +=
        `\t- **生產間隔**：[interval${DV} ${intervalSeconds}] 秒（[tickRate${DV} ${tickRate}] 黏液刻）\n`;
}

if (capacity !== null) {
    recipeOutput +=
        `\t- **儲電容量**：[capacity${DV} ${capacity}] J\n`;
}


// ======================================
// 合成材料
// ======================================
if (Object.keys(materials).length > 0) {

    recipeOutput += `\t**合成所需**：\n`;

    for (let matId in materials) {

        const qty = materials[matId];

        const realName =
            aliasToName[matId];

        const materialType =
            aliasToType[matId];

        if (realName) {

            recipeOutput +=
                `\t- [${materialType}${DV} [[${realName}]]] (qty${DV} ${qty})\n`;

        } else {

            recipeOutput +=
                `\t- [item${DV} ${matId}] (qty${DV} ${qty})\n`;
        }
    }

} else {

    recipeOutput +=
        `\t**合成所需**：擷取失敗\n`;
}


// 7. 根據 itemGroups.xxx 自動選擇建立位置
const folderDict = {
    "tools": "SF-工具(原版)",
    "weapons": "SF-武器(原版)",
    "basicMachines": "SF-基礎機器(原版)",
    "usefulItems": "SF-實用物品(原版)",
    "resources": "SF-礦物資源(原版)",
    "food": "SF-食物(原版)",

    // 暫時翻譯；若資料夾不存在會自動跳手動選擇
    "armor": "SF-裝備(原版)",
    "magicalArmor": "SF-魔法裝備(原版)",
    "magicalResources": "SF-魔法資源(原版)",
    "technicalComponents": "SF-科技元件(原版)",
    "technicalGadgets": "SF-科技工具(原版)",
    "magicalGadgets": "SF-魔法工具(原版)",
    "misc": "SF-雜項(原版)"
};

let folders = app.vault.getAllLoadedFiles().filter(
    f => f instanceof tp.obsidian.TFolder && f.path !== "/"
);

let folderPaths =
    folders.map(f => f.path);

// 例如 itemGroups.food → folderDict["food"]
let selectedPath =
    folderDict[itemGroupRaw];

let targetFolder =
    selectedPath
        ? folders.find(
            f => f.path === selectedPath
        )
        : null;


// DICT 沒列到，或翻譯後的資料夾實際不存在
// → 使用原本的 suggester 手動選擇
if (!targetFolder) {

    let promptText =
        itemGroupRaw
            ? `分類 ${itemGroupRaw} 尚未設定可用資料夾，請手動選擇...`
            : "未擷取到 itemGroup，請手動選擇建立位置...";

    selectedPath =
        await tp.system.suggester(
            folderPaths,
            folderPaths,
            false,
            promptText
        );

    if (!selectedPath) return;

    targetFolder =
        folders.find(
            f => f.path === selectedPath
        );
}

// 8. 準備檔案的初始內容
let outputValue = outputQty;
let typeValue =
    selectedPath === "MC-物品資料庫"
        ? "mc-item"
        : "sf-item";
let content = `---
type: ${typeValue}
id: ${itemId}
level: ${levelCost}
output: ${outputValue}
aliases: ${JSON.stringify([...itemAliases, itemId])}
---

#待補充

# 合成/取得

### 常規

${recipeOutput}

### 液體

### 無限附加

### 煉金術自傳-煉金術

# 用途

- 

# 產物

- 
`;
await tp.file.create_new(
    content,
    itemName,
    true,
    targetFolder
);
%>
