<%*

//- **生產設備**：(Machine:: [[合金冶煉機]])
//	**合成所需**：
//	- [item:: [[鋼錠]]] (qty:: 1)
//	- [item:: [[銅錠]]] (qty:: 1)
//1. 設計一個互動式表單，使用 suggester 和 prompt 功能
//2. 先問合成 dataname 所需材料種類
//3. 問生產設備，使用 suggester，並且該步驟可Esc跳過
//4. 問合成所需物品，前面所輸入的數量為問的次數
//5. 一樣使用 suggester 問種類，可跳過，IF 沒跳過則可以輸入數量
//6. 根據輸入的 生產設備及所需材料種類 輸出**生產設備**：(Machine:: [[合金冶煉機]]) 連結、以及相對應數量的材料

// ======================================
// 1. 取得當前檔案名稱
// ======================================
const dataname = tp.file.title;
const DV = "::";





// ======================================
// 3. 取得 Vault 內所有 Markdown 檔案
// ======================================
const files = app.vault.getMarkdownFiles();

const allFiles =
    files.map(f => f.basename);


// ======================================
// 4. 建立「檔名 → TYPE」字典
// ======================================
const typeDict = {
    "sf-item": "item",
    "mc-item": "item",
    "sf-GEO": "GEO",
    "sf-drop": "drop"
};

const fileTypeDict = {};

for (let f of files) {

    const frontmatter =
        app.metadataCache
            .getFileCache(f)
            ?.frontmatter || {};

    let fileType =
        frontmatter.type || "sf-item";

    fileTypeDict[f.basename] =
        typeDict[fileType] || "item";
}


// ======================================
// 5. 生產設備清單
// ======================================
const machines = [
    "冶煉機",
    "攪拌機",
    "魔法工作台",
    "強化工作台",
    "冶煉爐"
];

const machineOptions = [
    ...new Set([
        ...machines,
        ...allFiles
    ])
];


// 在第一個選項加入「手動輸入」
const machineDisplay = [
    "✏️ 手動輸入",
    ...machineOptions
];

const machineValues = [
    "__MANUAL__",
    ...machineOptions
];


// ======================================
// 6. 詢問生產設備
// ======================================
let machine =
    await tp.system.suggester(
        machineDisplay,
        machineValues,
        false,
        "請選擇「生產設備」(ESC 可跳過)"
    );

let machineIsManual = false;


// 選擇「手動輸入」
if (machine === "__MANUAL__") {

    machine =
        await tp.system.prompt(
            "請輸入生產設備名稱："
        );

    if (machine) {
        machine = machine.trim();
        machineIsManual = true;
    }
}

// ======================================
// 2. 詢問材料種類數量
// ======================================
let countStr = await tp.system.prompt(
    `請問合成「${dataname}」需要幾種材料？ (請輸入 1~9)`,
    "1"
);

let count = parseInt(countStr);


// 防呆：只能輸入 1~9
if (
    isNaN(count) ||
    count < 1 ||
    count > 9
) {
    new Notice(
        "❌ 腳本已取消：請輸入 1~9 的有效數字！"
    );
    return;
}

// ======================================
// 7. 材料 Suggester
// ======================================
const itemDisplay = [
    "✏️ 手動輸入",
    ...allFiles
];

const itemValues = [
    "__MANUAL__",
    ...allFiles
];


// ======================================
// 8. 逐項詢問材料
// ======================================
let ingredients = [];

for (let i = 0; i < count; i++) {

    let itemName =
        await tp.system.suggester(
            itemDisplay,
            itemValues,
            false,
            `【第 ${i + 1}/${count} 項材料】請選擇物品 (ESC 可跳過)`
        );

    let isManual = false;


    // ----------------------------------
    // 手動輸入不存在於 Vault 的材料
    // ----------------------------------
    if (itemName === "__MANUAL__") {

        itemName =
            await tp.system.prompt(
                `【第 ${i + 1}/${count} 項材料】請輸入物品名稱：`
            );

        if (itemName) {
            itemName = itemName.trim();
            isManual = true;
        }
    }


    // ----------------------------------
    // 有選到 / 有輸入材料才問數量
    // ESC 或取消 prompt 就跳過
    // ----------------------------------
    if (itemName) {

        let itemQtyStr =
            await tp.system.prompt(
                `請輸入「${itemName}」的所需數量：`,
                "1"
            );

        let itemQty =
            parseInt(itemQtyStr);

        // 數量輸入錯誤時預設 1
        if (
            isNaN(itemQty) ||
            itemQty < 1
        ) {
            itemQty = 1;
        }


        // 已存在 BIO → 根據 type
        // 手動輸入 → 暫時視為 item
        let itemType =
            isManual
                ? "item"
                : (fileTypeDict[itemName] || "item");


        ingredients.push({
            name: itemName,
            qty: itemQty,
            type: itemType,
            manual: isManual
        });
    }
}


// ======================================
// 9. 組合生產設備
// ======================================
let machineText = "無/手動";

if (machine) {

    // Suggester 選到既有檔案 → 建立連結
    if (
        !machineIsManual &&
        allFiles.includes(machine)
    ) {
        machineText =
            `[[${machine}]]`;
    }

    // 手動輸入或不是 Vault 檔案 → 純文字
    else {
        machineText = machine;
    }
}


// ======================================
// 10. 組合輸出
// ======================================
let output =
    `- **生產設備**：(Machine${DV} ${machineText})\n`;


// 有材料才輸出「合成所需」
if (ingredients.length > 0) {

    output +=
        `\t**合成所需**：\n`;

    for (let ing of ingredients) {

        // Vault 已存在 → Wikilink
        if (!ing.manual) {

            output +=
                `\t- [${ing.type}${DV} [[${ing.name}]]] (qty${DV} ${ing.qty})\n`;
        }

        // 手動輸入 → 純文字
        else {

            output +=
                `\t- [item${DV} ${ing.name}] (qty${DV} ${ing.qty})\n`;
        }
    }
}


// ======================================
// 11. 輸出到目前筆記
// ======================================
tR += output;
_%>
