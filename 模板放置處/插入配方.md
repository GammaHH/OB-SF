<%#
- **生產設備**：(Machine:: [[合金冶煉機]])
	**合成所需**：
	- [item:: [[鋼錠]]] (qty:: 1)
	- [item:: [[銅錠]]] (qty:: 1)

1. 設計一個互動式表單，使用 suggester 和 prompt 功能
2. 先問合成 `<%`dataname`%>` 所需材料種類 (只能輸入 1~9)
3. 問生產設備，使用 suggester，並且該步驟可Esc跳過
4. 問合成所需物品，前面所輸入的數量為問的次數
5. 一樣使用 suggester 問種類，可跳過，IF 沒跳過則可以輸入數量
6. 根據輸入的 生產設備及所需材料種類 輸出**生產設備**：(Machine:: [[合金冶煉機]]) 連結、以及相對應數量的材料
_%>

<%*
// 1. 取得當前檔案名稱 (作為提示詞使用)
const dataname = tp.file.title;

// 2. 詢問合成所需的材料種類數量 (只能輸入 1~9)
let countStr = await tp.system.prompt(`請問合成「${dataname}」需要幾種材料？ (請輸入 1~9)`, "1");
let count = parseInt(countStr);

// 防呆機制：確保輸入的是 1~9 的數字，否則中斷執行
if (isNaN(count) || count < 1 || count > 9) {
    new Notice("❌ 腳本已取消：請輸入 1~9 的有效數字！");
    return;
}

// 取得庫內所有檔案清單供 Suggester 搜尋
const allFiles = app.vault.getMarkdownFiles().map(f => f.basename);

// (可選) 建立常用的機器清單，放在選單最前面方便點選，找不到依然可以搜尋全庫檔案
const machines = ["合金冶煉機", "攪拌機", "魔法工作台", "強化工作台", "冶煉爐"];
const machineOptions = [...new Set([...machines, ...allFiles])]; 

// 3. 詢問生產設備 (按 ESC 會回傳 null 或 undefined)
let machine = await tp.system.suggester(
    machineOptions, 
    machineOptions, 
    false, 
    "請選擇「生產設備」(按 ESC 可跳過)"
);

// 4 & 5. 根據輸入的次數，迴圈收集材料種類與數量
let ingredients = [];
for (let i = 0; i < count; i++) {
    // 詢問材料種類 (按 ESC 跳過)
    let itemName = await tp.system.suggester(
        allFiles, 
        allFiles, 
        false, 
        `【第 ${i+1}/${count} 項材料】請選擇物品 (按 ESC 可跳過)`
    );
    
    // IF 沒跳過則可以輸入數量
    if (itemName) {
        let itemQtyStr = await tp.system.prompt(`請輸入「${itemName}」的所需數量：`, "1");
        let itemQty = parseInt(itemQtyStr) || 1; // 防呆，若未輸入則預設為 1
        ingredients.push({ name: itemName, qty: itemQty });
    }
}

// 6. 組合輸出字串 (使用 \t 縮排來精準達成你要求的巢狀結構)
let machineText = machine ? `[[${machine}]]` : "無/手動";
let output = `- **生產設備**：(Machine:: ${machineText})\n\t**合成所需**：\n`;

for (let ing of ingredients) {
    output += `\t- [item:: [[${ing.name}]]] (qty:: ${ing.qty})\n`;
}

// 將最終字串印在筆記上
tR += output;
_%>
