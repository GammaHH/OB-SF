<%*
// ========================================
// 1. 基本資料輸入
// ========================================
let rawInput = await tp.system.prompt(
    "請輸入新物品名稱 (若有別名請用空格隔開)："
);
let itemId = await tp.system.prompt(
    "請輸入物品 ID (例如: FEIXIANXINGSW)："
);

// 防呆
if (!rawInput || !itemId) return;

let nameParts = rawInput.trim().split(/\s+/);
let itemName = nameParts[0];
let itemAliases = nameParts.slice(1);


// ========================================
// 2. 基本變數
// ========================================
let itemSource = "";
let entity = "";
let chance = "";
let levelCost = "";

const referenceFolder = "參考資料";


// ========================================
// 3. 搜尋該 ID 的原始碼區塊 + 解鎖等級
// ========================================
try {

    const list = await app.vault.adapter.list(referenceFolder);

    for (let filePath of list.files) {

        if (
            filePath.endsWith(".yml") ||
            filePath.endsWith(".txt") ||
            filePath.endsWith(".md")
        ) {

            let sourceContent =
                await app.vault.adapter.read(filePath);


            // ======================================
            // A. 找解鎖等級
            // ======================================
            if (!levelCost) {

                let blocks =
                    sourceContent.split(/\n(?=[a-zA-Z0-9_]+:)/);

                for (let block of blocks) {

                    let idRegex =
                        new RegExp(`-\\s*${itemId}\\b`);

                    if (idRegex.test(block)) {

                        let levelMatch =
                            block.match(/levelCost:\s*(\d+)/);

                        if (levelMatch) {
                            levelCost = levelMatch[1];
                        }

                        break;
                    }
                }
            }


            // ======================================
            // B. 找物品原始碼
            // ======================================
            if (!itemSource) {

                let escapedItemId =
                    itemId.replace(
                        /[.*+?^${}()|[\]\\]/g,
                        "\\$&"
                    );

                let blockRegex =
                    new RegExp(
                        `^${escapedItemId}:[\\s\\S]*?(?=^\\S[^:\\r\\n]|$(?![\\s\\S]))`,
                        "gm"
                    );

                let blockMatch =
                    sourceContent.match(blockRegex);

                if (blockMatch) {
                    itemSource = blockMatch[0];
                }
            }


            // 兩者都有找到才停止
            if (levelCost && itemSource) {
                break;
            }
        }
    }

} catch (error) {

    new Notice(
        `⚠️ 搜尋參考資料時發生錯誤：${error.message}`
    );

    return;
}


// ========================================
// 4. 解析 itemSource
// ========================================
if (itemSource) {

    // ------------------------------------
    // A. YAML 名稱 → 加入 aliases
    // ------------------------------------
    let nameMatch = itemSource.match(
        /name:\s*['"]?([^'"\n]+)['"]?/
    );

    if (nameMatch) {

        let yamlName = nameMatch[1]
            .replace(/&[0-9a-fk-orlmn]/ig, "")
            .trim();

        if (
            yamlName !== itemName &&
            !itemAliases.includes(yamlName)
        ) {
            itemAliases.push(yamlName);
        }
    }


    // ------------------------------------
    // B. 生物掉落來源
    // ------------------------------------
    let entityMatch = itemSource.match(
        /^[ \t]*entity:\s*([A-Za-z0-9_:-]+)/m
    );

    if (entityMatch) {
        entity = entityMatch[1];
    }


    // ------------------------------------
    // C. 掉落機率
    // ------------------------------------
    let chanceMatch = itemSource.match(
        /^[ \t]*chance:\s*([\d.]+)/m
    );

    if (chanceMatch) {
        chance = chanceMatch[1];
    }
}


// ========================================
// 5. 建立取得方式輸出
// ========================================
let recipeOutput = "";
const dv = "::";
if (entity) {
    recipeOutput += `- **取得方式**：生物掉落\n`;
    recipeOutput += `\t- **來源生物**：[entity${dv} ${entity}]\n`;

    if (chance) {
        recipeOutput += `\t- **掉落機率**：[chance${dv} ${chance}]%\n`;
    } else {
        recipeOutput += `\t- **掉落機率**：(chance${dv} 待補充)\n`;
    }
}


// ========================================
// 6. ID 前綴 → 預設資料夾
// ========================================
const folderDict = {
    "HM": "SF-海曼科技院",
    "SF": "SF-黏液科技",
    "SC": "SF-科技院",
    "GN": "SF-基因科技"
};


// ========================================
// 7. 決定建立資料夾
// ========================================
let folders = app.vault
    .getAllLoadedFiles()
    .filter(
        f =>
            f instanceof tp.obsidian.TFolder &&
            f.path !== "/"
    );

let folderPaths =
    folders.map(f => f.path);


// HM_WJXXJCY → HM
let idPrefix =
    itemId.split("_")[0];


// 先從字典找
let selectedPath =
    folderDict[idPrefix];


// 找不到才手動選
if (!selectedPath) {

    selectedPath =
        await tp.system.suggester(
            folderPaths,
            folderPaths,
            false,
            "請選擇要將此物品創建在哪個資料夾..."
        );

    if (!selectedPath) return;
}


let targetFolder =
    folders.find(
        f => f.path === selectedPath
    );


// 如果字典寫錯資料夾名稱
if (!targetFolder) {

    new Notice(
        `❌ 找不到資料夾：${selectedPath}`
    );

    return;
}


// ========================================
// 8. 建立 BIO 內容
// ========================================
let content = `---
type: sf-drop
id: ${itemId}
level: ${levelCost}
aliases: ${JSON.stringify([...itemAliases, itemId])}
---

#待補充

# 合成/取得

### 常規

${recipeOutput}

# 用途

-

# 產物

-
`;


// ========================================
// 9. 建立檔案
// ========================================
await tp.file.create_new(
    content,
    itemName,
    true,
    targetFolder
);

%>
