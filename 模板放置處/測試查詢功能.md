
<%*
// 1. 詢問要測試的物品 ID
let itemId = await tp.system.prompt("請輸入要測試的物品 ID (例如: HM_QKJCY)：");
if (!itemId) return;

let levelCost = "無";
let itemSource = "";
const referenceFolder = "參考資料";

try {
    const list = await app.vault.adapter.list(referenceFolder);
    for (let filePath of list.files) {
        if (filePath.endsWith(".yml") || filePath.endsWith(".txt") || filePath.endsWith(".md")) {
            let content = await app.vault.adapter.read(filePath);
            
            // 任務 A: 找解鎖等級
            if (levelCost === "無") {
                let blocks = content.split(/\n(?=[a-zA-Z0-9_]+:)/);
                for (let block of blocks) {
                    let idRegex = new RegExp(`-\\s*${itemId}\\b`);
                    if (idRegex.test(block)) {
                        let levelMatch = block.match(/levelCost:\s*(\d+)/);
                        if (levelMatch) levelCost = levelMatch[1];
                        break;
                    }
                }
            }
            
            // 任務 B: 找配方源碼
            if (!itemSource) {
                let blockRegex = new RegExp(`^${itemId}:[\\s\\S]*?(?=^\\S[^:\\r\\n]|$(?![\\s\\S]))`, "gm");
                let blockMatch = content.match(blockRegex);
                if (blockMatch) {
                    // 因為這裡是用整個匹配結果 (blockMatch[0])，而不是群組，所以改抓 [0]
                    itemSource = blockMatch[0];
                }
            }
            
            if (levelCost !== "無" && itemSource) break;
        }
    }
} catch (error) {
    new Notice("⚠️ 找不到參考資料，請確認資料夾名稱或路徑是否正確！");
    return;
}
// ==========================================
// 3. 解析取得的配方源碼
// ==========================================
let machine = "無/手動";
let materials = {};

if (itemSource) {
    // 設備翻譯字典
    const machineDict = {
        "ENHANCED_CRAFTING_TABLE": "強化工作台",
        "MAGIC_WORKBENCH": "魔法工作台",
        "ARMOR_FORGE": "護甲鍛造台",
        "SMELTERY": "冶煉爐",
        "ORE_CRUSHER": "碎礦機",
        "COMPRESSOR": "壓縮機",
        "JUICER": "榨汁機",
        "ANCIENT_ALTAR": "遠古祭壇",
        "HEATED_PRESSURE_CHAMBER": "加熱壓力艙"
    };
	machineRaw = itemSource.match(/(?<=recipe_type:\s)[A-Za-z0-9_]+/)?.[0] || "無/手動";
	machine = machineDict[machineRaw] || machineRaw;
	
    // ------------------------------
    // B. 找 recipe:
    // ------------------------------
    let recipeMatch = itemSource.match(/^[ \t]*recipe:\s*$/m);

    if (recipeMatch) {

        // recipe: 後面的全部內容
        let recipeSource = itemSource.slice(
            recipeMatch.index + recipeMatch[0].length
        );

        // ------------------------------
        // C. 抓 material + amount
        // ------------------------------
        const ingredientRegex =
            /^[ \t]+material:\s*([A-Za-z0-9_:-]+)\s*\r?\n[ \t]+amount:\s*(\d+)/gm;

        for (let match of recipeSource.matchAll(ingredientRegex)) {

            let material = match[1];
            let amount = Number(match[2]);

            // 相同材料累加
            if (!materials[material]) {
                materials[material] = 0;
            }

            materials[material] += amount;
        }
    }
}

let materialOutput = "";

if (Object.keys(materials).length > 0) {
    materialOutput = Object.entries(materials)
        .map(([material, amount]) => `- ${material} × ${amount}`)
        .join("\n");
} else {
    materialOutput = "- 無";
}

let output = `### 測試結果 - ${itemId}

- **所需等級**：${levelCost}
- **製作設備**：${machine}

### 所需材料

${materialOutput}

### 抓取到的源碼

\`\`\`yaml
${itemSource}
\`\`\`
`;

tR += output;
_%>
