<%*
// 取得當前活躍的編輯器視窗
const view = app.workspace.getActiveViewOfType(tp.obsidian.MarkdownView);

if (view) {
    const editor = view.editor;
    let text = editor.getValue();

    // 1. 將所有的 #### 替換為 ###
    text = text.replace(/####/g, "###");

    // 2. 將所有的半形冒號 : 替換為全形冒號 ：
    text = text.replace(/:/g, "：");

    // 將修改後的文字重新寫入編輯器
    editor.setValue(text);
}
%>
