```dataviewjs
const path = "git-history.txt";

try {
    const text = await app.vault.adapter.read(path);

    const commits = text
        .split(/(?=^@@@)/m)
        .filter(x => x.trim().startsWith("@@@"))
        .map(chunk => {

            const header = chunk
                .split(/\r?\n/)[0]
                .replace(/^@@@/, "");

            const [date, hash, ...message] = header.split("|");

            const added = Number(
                chunk.match(/(\d+) insertions?\(\+\)/)?.[1] ?? 0
            );

            const deleted = Number(
                chunk.match(/(\d+) deletions?\(-\)/)?.[1] ?? 0
            );

            return {
                date,
                hash,
                message: message.join("|"),
                added,
                deleted
            };
        });

    dv.header(2, "OB-SF 開發紀錄");

    dv.paragraph(`📦 Commits：${commits.length}`);

    dv.table(
        ["日期", "Commit", "新增行數", "刪除行數", "開發內容"],
        commits.map(c => [
            c.date,
            c.hash,
            `+${c.added}`,
            `−${c.deleted}`,
            c.message
        ])
    );

} catch (error) {
    dv.paragraph("⚠ 讀取失敗：" + error.message);
}
```
